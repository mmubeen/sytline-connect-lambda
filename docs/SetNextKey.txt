ALTER PROCEDURE [dbo].[SetNextKeySp] (
  @TableName  sysname
, @ColumnName sysname
, @Prefix     LongList
, @KeyLength  INT
, @Type       LongList
, @SubKey     GenericKeyType = NULL
, @TableName2 sysname        = NULL
, @SubKey2    GenericKeyType = NULL
, @KeyStr     LongList OUTPUT
, @Infobar    Infobar  OUTPUT
, @Increment  INT = 1
)
AS

   -- Check for existence of Generic External Touch Point routine (this section was generated by SpETPCodeSp and inserted by CallETPs.exe):
   IF OBJECT_ID(N'dbo.EXTGEN_SetNextKeySp') IS NOT NULL
   BEGIN
      DECLARE @EXTGEN_SpName sysname
      SET @EXTGEN_SpName = N'dbo.EXTGEN_SetNextKeySp'
      -- Invoke the ETP routine, passing in (and out) this routine's parameters:
      DECLARE @EXTGEN_Severity int
      EXEC @EXTGEN_Severity = @EXTGEN_SpName
         @TableName
         , @ColumnName
         , @Prefix
         , @KeyLength
         , @Type
         , @SubKey
         , @TableName2
         , @SubKey2
         , @KeyStr OUTPUT
         , @Infobar OUTPUT
         , @Increment
 
      -- ETP routine can RETURN 1 to signal that the remainder of this standard routine should now proceed:
      IF @EXTGEN_Severity <> 1
         RETURN @EXTGEN_Severity
   END
   -- End of Generic External Touch Point code.
 
DECLARE
  @Severity INT
, @KeyNum   DECIMAL(25, 0) -- room for big integers
, @KeyNum2  DECIMAL(25, 0) -- room for big integers
, @i        INT
, @KeyVal   LongList
, @KeyVal2  LongList
, @Fill     NCHAR
, @MsgParam1 DefaultCharType

SET @Infobar = NULL
SET @Prefix = ISNULL (@Prefix, '')

-- Remove "0" as a valid prefix

IF @Prefix NOT LIKE '%[^0-9]%'
    IF CAST (@Prefix AS INT) = 0
        SET @Prefix = ''

-- Remove spaces as a valid prefix
IF LEN(REPLACE(@Prefix, ' ', '')) = 0
    SET @Prefix = ''

EXEC @Severity = Next2SubKeySp
  @TableName  = @TableName
, @ColumnName = @ColumnName
, @Prefix     = @Prefix
, @KeyLength  = @KeyLength
, @SubKey     = @SubKey
, @Increment  = @Increment
, @KeyVal     = @KeyVal  OUTPUT
, @Infobar    = @Infobar OUTPUT

IF @Severity <> 0
    RETURN @Severity

SET @i = ISNULL (LEN(@Prefix), 0)

IF @KeyVal IS NOT NULL AND @KeyVal LIKE replace(@Prefix, '[', '[[]') + '%'
BEGIN
    -- Commented out the following checks 06/18/02
    --IF dbo.can-do('call-avail', 'extfin') = 1 AND
    --  dbo.can-do('call-avail', 'ext-multi-curr') = 1 AND
    IF @Type IN ('CUST', 'VEND')
        SET @KeyNum = CAST (
          SUBSTRING (@KeyVal, LEN(@Prefix) + 1, @KeyLength - @i) AS DECIMAL(25, 0))
    ELSE
        SET @KeyNum = CAST (
          SUBSTRING(@KeyVal, LEN(@Prefix) + 1, 4000) AS DECIMAL(25, 0))
END -- Keyval found

IF @TableName2 IS NOT NULL
BEGIN
    EXEC @Severity = Next2SubKeySp
      @TableName  = @TableName2
    , @ColumnName = @ColumnName
    , @Prefix     = @Prefix
    , @KeyLength  = @KeyLength
    , @SubKey     = @SubKey2
    , @Increment  = @Increment
    , @KeyVal     = @KeyVal2 OUTPUT
    , @Infobar    = @Infobar OUTPUT

    IF @Severity <> 0
        RETURN @Severity

END -- @TableName2 IS NOT NULL

IF @KeyVal2 IS NOT NULL AND @KeyVal2 LIKE replace(@Prefix, '[', '[[]') + '%'
BEGIN
    -- Commented out the following checks 06/18/02
    --IF dbo.can-do('call-avail', 'extfin') = 1 AND
    --  dbo.can-do('call-avail', 'ext-multi-curr') = 1 AND
    IF @Type IN ('CUST', 'VEND')
        SET @KeyNum2 = CAST (
          SUBSTRING (@KeyVal2, LEN(@Prefix) + 1, @KeyLength - @i) AS DECIMAL(25, 0))
    ELSE
        SET @KeyNum2 = CAST (
          SUBSTRING(@KeyVal2, LEN(@Prefix) + 1, 4000) AS DECIMAL(25, 0))
END -- Keyval found

IF ISNULL (@KeyNum2, -1) > ISNULL (@KeyNum, 0)
    SET @KeyNum = @KeyNum2

SET @KeyNum = ISNULL(@KeyNum, 0)

SET @KeyStr = @KeyNum
IF LEN(@Prefix) > 0
    SET @Fill = '0'
ELSE
    SET @Fill = ' '

SET @KeyStr = @Prefix +
   REPLICATE (@Fill, CASE WHEN ( @KeyLength - LEN(@Prefix) - LEN(@KeyStr) ) < 0 THEN 0 ELSE ( @KeyLength - LEN(@Prefix) - LEN(@KeyStr)) END ) + @KeyStr

SET @MsgParam1 = '@<TableName>.<ColumnName>'
SET @MsgParam1 = REPLACE(@MsgParam1, '<TableName>', @TableName)
SET @MsgParam1 = REPLACE(@MsgParam1, '<ColumnName>', @ColumnName)

IF LEN(@KeyStr) > @KeyLength
BEGIN
    EXEC @Severity = MsgAppSp @Infobar OUTPUT
     , 'E=NoCompare', @MsgParam1, @KeyStr
    EXEC @Severity = MsgAppSp @Infobar OUTPUT
     , 'E=NoCompareLen>', @MsgParam1, @KeyLength
    RETURN @Severity
END

RETURN 0
