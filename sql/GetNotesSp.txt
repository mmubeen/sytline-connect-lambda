@ObjectName SYSNAME,
@RefRowPointer RowPointertype
AS

      -- Invoke the ETP routine, passing in (and out) this routine's parameters:
      EXEC @EXTGEN_SpName
         @ObjectName
         , @RefRowPointer
 
      -- ETP routine must take over all desired functionality of this standard routine:
      RETURN 0
   END
   -- End of Generic External Touch Point code.
 
SELECT NoteContent,NoteDesc
FROM SpecificNotes spcn,NoteHeaders nh,ObjectNotes objn

        objn.RefRowPointer = @RefRowPointer
