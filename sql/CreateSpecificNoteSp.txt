ALTER PROCEDURE [dbo].[CreateSpecificNoteSp] (
  @ObjectName   SYSNAME
, @RowPointer   RowPointerType
, @NoteDesc     LongDescType
, @NoteText     LongListType
, @InternalFlag FlagIeType
, @Infobar      Infobar OUTPUT
)
AS

   -- Check for existence of Generic External Touch Point routine (this section was generated by SpETPCodeSp and inserted by CallETPs.exe):
   IF OBJECT_ID(N'dbo.EXTGEN_CreateSpecificNoteSp') IS NOT NULL
   BEGIN
      DECLARE @EXTGEN_SpName sysname
      SET @EXTGEN_SpName = N'dbo.EXTGEN_CreateSpecificNoteSp'
      -- Invoke the ETP routine, passing in (and out) this routine's parameters:
      DECLARE @EXTGEN_Severity int
      EXEC @EXTGEN_Severity = @EXTGEN_SpName
         @ObjectName
         , @RowPointer
         , @NoteDesc
         , @NoteText
         , @InternalFlag
         , @Infobar OUTPUT
 
      -- ETP routine can RETURN 1 to signal that the remainder of this standard routine should now proceed:
      IF @EXTGEN_Severity <> 1
         RETURN @EXTGEN_Severity
   END
   -- End of Generic External Touch Point code.
 
DECLARE
  @SpecificNoteToken TokenType
, @Severity          INT

SET @Severity = 0

-- Signal SpecificNotesInsert to store the generated Identity value in a Process Variable:
EXEC dbo.AnticipateSessionIdentitySp N'SpecificNotes'

INSERT INTO SpecificNotes (
  NoteContent
, NoteDesc
) VALUES (
  @NoteText
, @NoteDesc
)

EXEC dbo.RetrieveSessionIdentitySp N'SpecificNotes', @SpecificNoteToken OUTPUT

EXEC @Severity = AttachSpecificNoteSp
  @TableName  = @ObjectName
, @RowPointer = @RowPointer
, @NoteToken  = @SpecificNoteToken
, @NoteFlag   = @InternalFlag
, @NoteType   = 0

RETURN @Severity
